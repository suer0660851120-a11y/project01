pythonfrom flask import Flask, request, jsonify
from flask_cors import CORS
import requests

app = Flask(__name__)
CORS(app)  # อนุญาตให้เว็บเรียก API ได้

# ใส่ Token ของแต่ละสาขา
LINE_TOKENS = {
    "สาขา 1 - สยาม":      "TOKEN_สาขา1_ใส่ตรงนี้",
    "สาขา 2 - อโศก":      "TOKEN_สาขา2_ใส่ตรงนี้",
    "สาขา 3 - ลาดพร้าว":  "TOKEN_สาขา3_ใส่ตรงนี้",
}

def send_line_notify(token, message):
    url = "https://notify-api.line.me/api/notify"
    headers = {"Authorization": f"Bearer {token}"}
    data = {"message": message}
    response = requests.post(url, headers=headers, data=data)
    return response.status_code

@app.route("/api/notify", methods=["POST"])
def notify():
    data = request.get_json()

    branch      = data.get("branch")
    queue_number = data.get("queueNumber")
    name        = data.get("name")

    # เลือก Token ตามสาขา
    token = LINE_TOKENS.get(branch)
    if not token:
        return jsonify({"error": "ไม่พบสาขา"}), 400

    # สร้างข้อความแจ้งเตือน
    message = f"""
🔔 มีคิวใหม่!
━━━━━━━━━━━━
สาขา     : {branch}
หมายเลขคิว: #{queue_number}
ชื่อลูกค้า : {name}
━━━━━━━━━━━━
"""
    status = send_line_notify(token, message)

    if status == 200:
        return jsonify({"success": True})
    else:
        return jsonify({"error": "ส่ง LINE ไม่สำเร็จ"}), 500

if __name__ == "__main__":
    app.run(debug=True)
